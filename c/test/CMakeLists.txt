# bz2-ng tests.  Everything here generates its data in code (bzip2's own
# sample files in reference/ excepted); the corpora under ../data are used by
# scripts/bench_run.sh and can be given to dec_difftest as extra inputs.

# The randomising compressor: libbz2's compress.c (reference/, unmodified)
# with two changes applied at configure time, so that the decoder's
# derandomisation path can be tested: each block is randomised before it is
# sorted (rand_hook.c), and its randomised bit is set.  Built with every
# symbol renamed rand_*.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/reference/compress.c _compress)
# Replaces the one occurrence of from by to in the variable _compress.
function(bzng_patch_compress from to)
  string(FIND "${_compress}" "${from}" _first)
  string(FIND "${_compress}" "${from}" _last REVERSE)
  if(_first EQUAL -1 OR NOT _first EQUAL _last)
    message(FATAL_ERROR "reference/compress.c: a patch for the randomising compressor does not apply")
  endif()
  string(REPLACE "${from}" "${to}" _patched "${_compress}")
  set(_compress "${_patched}" PARENT_SCOPE)
endfunction()
bzng_patch_compress("#include \"bzlib_private.h\"\n"
                    "#include \"bzlib_private.h\"\nvoid bzng_test_randomise_block ( EState* s );\n")
bzng_patch_compress("      BZ2_blockSort ( s );"
                    "      bzng_test_randomise_block ( s );\n      BZ2_blockSort ( s );")
bzng_patch_compress("      bsW(s,1,0);\n\n      bsW ( s, 24, s->origPtr );"
                    "      bsW(s,1,1); /* randomised */\n\n      bsW ( s, 24, s->origPtr );")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/rand_compress.c
     "/* Generated by test/CMakeLists.txt from reference/compress.c (bzip2 1.0.8,\n"
     "   licence in reference/LICENSE): ALTERED to write randomised blocks. */\n"
     "${_compress}")
set(_rand_srcs ${CMAKE_CURRENT_BINARY_DIR}/rand_compress.c rand_hook.c)
foreach(_f IN LISTS BZ2NG_REFERENCE_SOURCES)
  if(NOT _f MATCHES "/compress[.]c$")
    list(APPEND _rand_srcs ${PROJECT_SOURCE_DIR}/${_f})
  endif()
endforeach()
add_library(bz2_rand STATIC ${_rand_srcs})
target_include_directories(bz2_rand BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/reference)
target_compile_definitions(bz2_rand PRIVATE BZPFX=rand_ _FILE_OFFSET_BITS=64 _DEFAULT_SOURCE)
target_compile_options(bz2_rand PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/reference_rename.h -w)

# bz2-ng with each decoder memory layout as BZ2_bzDecompressInit's default,
# so that every test that reaches the decoder through the public API runs on
# both layouts, whichever one the library is configured with.
foreach(_layout FIT WIDE)
  string(TOLOWER ${_layout} _name)
  add_library(bz2_layout_${_name} STATIC $<TARGET_OBJECTS:bz2_core_objects> ${PROJECT_SOURCE_DIR}/lib/decompress.c)
  bzng_library_settings(bz2_layout_${_name})
  target_compile_definitions(bz2_layout_${_name} PRIVATE BZNG_DEFAULT_LAYOUT=BZNG_LAYOUT_${_layout})
endforeach()

# A test program linked against bz2-ng (static, so internal functions such as
# bzng_decompress_init and bzng_tier_name are reachable) and the reference.
function(bzng_test_program name)
  add_executable(${name} ${ARGN})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/lib)
  target_compile_options(${name} PRIVATE ${BZ2NG_WARNINGS})
  target_compile_definitions(${name} PRIVATE _DEFAULT_SOURCE _DARWIN_C_SOURCE _FILE_OFFSET_BITS=64)
  target_link_libraries(${name} PRIVATE bz2_static bz2_reference bz2_rand)
endfunction()

# The kernel tiers this machine can run, for tests run once per tier.

  set(BZ2NG_TEST_TIERS scalar sse4 avx2)

  set(BZ2NG_TEST_TIERS scalar)
endif()

# A test run once per tier with BZ2NG_TIER set; the program reports skipped
# (exit 77) when the CPU lacks the tier (test_tier.h).
function(bzng_tier_test name)
  foreach(_tier IN LISTS BZ2NG_TEST_TIERS)
    add_test(NAME ${name}_${_tier} COMMAND ${ARGN})
    set_tests_properties(${name}_${_tier} PROPERTIES SKIP_RETURN_CODE 77 ENVIRONMENT "BZ2NG_TIER=${_tier}")
  endforeach()
endfunction()

# The sorter's kernel forms (lib/sort.c) follow the compiler and target, so a
# build of the library compiles one combination of them.  The oracle is also
# built with its own lib/sort.c in each combination that some build ships,
# which bz2_static's copy then gives way to at link time, so that every
# machine tests all of them (apple_*: Apple silicon).  The sorter uses no
# kernel tier, so each runs once.
function(bzng_sort_form_test name mask_selects skip_empty_slots suffix0_apart branchless_naming)
  bzng_test_program(sort_test_form_${name} sort_test.c ${PROJECT_SOURCE_DIR}/lib/sort.c)
  target_compile_definitions(sort_test_form_${name} PRIVATE
    SORT_MASK_SELECTS=${mask_selects} SORT_SKIP_EMPTY_SLOTS=${skip_empty_slots}
    SORT_SUFFIX0_APART=${suffix0_apart} SORT_BRANCHLESS_NAMING=${branchless_naming})
  add_test(NAME sort_oracle_form_${name} COMMAND sort_test_form_${name})
endfunction()
bzng_sort_form_test(apple_clang 0 0 0 1)
bzng_sort_form_test(clang 0 0 1 0)
bzng_sort_form_test(gcc 1 1 1 0)
bzng_sort_form_test(apple_gcc 1 1 1 1)
